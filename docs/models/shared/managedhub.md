# ManagedHub

## Example Usage

```typescript
import { HomeVisibility, ManagedHub, RecommendationsVisibility } from "@parke.dev/plexjs/models/shared";

let value: ManagedHub = {
  title: "string value",
  homeVisibility: HomeVisibility.All,
  identifier: "string value",
  promotedToOwnHome: true,
  promotedToRecommended: true,
  promotedToSharedHome: true,
  recommendationsVisibility: RecommendationsVisibility.All,
};
```
